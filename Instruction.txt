Create test data file:
$nx=64; $ny=64; $nz=64
$n = $nx*$ny*$nz
$bytes = New-Object byte[] ($n*4)

for($k=0;$k -lt $nz;$k++){
  for($j=0;$j -lt $ny;$j++){
    for($i=0;$i -lt $nx;$i++){
      $idx = ($k*$ny*$nx + $j*$nx + $i)
      $v = [single]([math]::Sin($i/7.0)+[math]::Cos($j/9.0)+$k/50.0)
      $b = [System.BitConverter]::GetBytes($v)
      $off = $idx*4
      $bytes[$off+0]=$b[0]; $bytes[$off+1]=$b[1]; $bytes[$off+2]=$b[2]; $bytes[$off+3]=$b[3]
    }
  }
}

Automatically handle the last chunk size too
This version reads the file size and computes the frame count automatically.
Since each value is float32, each voxel is 4 bytes.
For a chunk shaped:

cd C:\dev\SPERR\build_vs2022\bin\Release

$chunkDir = "C:\dev\SPERR\python_files\chunks"
$height = 270
$width = 480
$bytesPerValue = 4

Get-ChildItem $chunkDir -Filter *.raw | ForEach-Object {
    $in = $_.FullName
    $stream = [System.IO.Path]::ChangeExtension($in, ".stream")

    $frames = [int]($_.Length / ($height * $width * $bytesPerValue))

    Write-Host "Compressing $in"
    Write-Host "  dims = $frames $height $width"
    Write-Host "  out  = $stream"

    .\sperr3d.exe -c --ftype 32 --dims $frames $height $width --pwe 1e-4 --bitstream $stream $in

    if ($LASTEXITCODE -ne 0) {
        Write-Host "Failed on $in with exit code $LASTEXITCODE"
    }
}
